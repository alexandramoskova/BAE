        /* -------------------------------------------------------------
           1. AUDIO SYNTHESIZER (Shape of My Heart Acoustic Style)

        let audioCtx = null;
        let isPlaying = false;
        let audioInterval = null;

        const melodyNotes = [
            261.63, 293.66, 329.63, 349.23, 392.00, 440.00, 493.88, 523.25
        ];

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playAcousticNote(freq, duration = 1.2) {
            if (!audioCtx || audioCtx.state !== 'running') return;

            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();

            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);

            osc.connect(gain);
            gain.connect(audioCtx.destination);

            osc.start();
            osc.stop(audioCtx.currentTime + duration);
        }

        function startRomanticMelody() {
            let noteIdx = 0;
            const arpeggioPattern = [329.63, 392.00, 440.00, 493.88, 523.25, 493.88, 440.00, 392.00];
            
            audioInterval = setInterval(() => {
                if (isPlaying && audioCtx && audioCtx.state === 'running') {
                    playAcousticNote(arpeggioPattern[noteIdx % arpeggioPattern.length], 1.5);
                    noteIdx++;
                }
            }, 600);
        }

        function toggleAudio() {
            initAudio();
            const icon = document.getElementById('musicIcon');
            const text = document.getElementById('musicText');

            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }

            if (!isPlaying) {
                isPlaying = true;
                icon.className = "fas fa-music animate-spin text-lg";
                text.innerText = "Shape of My Heart (Playing)";
                startRomanticMelody();
            } else {
                isPlaying = false;
                clearInterval(audioInterval);
                icon.className = "fas fa-volume-mute text-lg";
                text.innerText = "Musik Di-pause";
            }
        }

        function handleAnswer(isCorrect, feedbackMessage) {
            const feedbackEl = document.getElementById('quizFeedback');
            feedbackEl.classList.remove('hidden', 'bg-emerald-100', 'text-emerald-800', 'bg-rose-100', 'text-rose-800');

            if (isCorrect) {
                feedbackEl.classList.add('bg-emerald-100', 'text-emerald-800');
                feedbackEl.innerHTML = `<i class="fas fa-check-circle mr-1"></i> ${feedbackMessage}`;
                feedbackEl.classList.remove('hidden');

                setTimeout(() => {
                    feedbackEl.classList.add('hidden');
                    currentQuizIndex++;
                    if (currentQuizIndex < quizData.length) {
                        loadQuizStep();
                    } else {
                        // Quiz Finished -> Transition to Envelope
                        document.getElementById('quizSection').classList.add('hidden');
                        document.getElementById('envelopeSection').classList.remove('hidden');
                    }
                }, 1400);
            } else {
                feedbackEl.classList.add('bg-rose-100', 'text-rose-800');
                feedbackEl.innerHTML = `<i class="fas fa-times-circle mr-1"></i> Opss, coba tebak lagi viacu my sweetheart! ??`;
                feedbackEl.classList.remove('hidden');
            }
        }

        /* -------------------------------------------------------------
           5. FALLING PETALS HTML5 CANVAS ANIMATION
           ------------------------------------------------------------- */
        const canvas = document.getElementById('petalCanvas');
        const ctx = canvas.getContext('2d');

        let width = canvas.width = window.innerWidth;
        let height = canvas.height = window.innerHeight;

        window.addEventListener('resize', () => {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        });

        class Petal {
            constructor() {
                this.reset();
            }

            reset() {
                this.x = Math.random() * width;
                this.y = Math.random() * -height;
                this.size = Math.random() * 12 + 8;
                this.speedY = Math.random() * 1.5 + 0.8;
                this.speedX = Math.random() * 1 - 0.5;
                this.rotation = Math.random() * 360;
                this.rotationSpeed = Math.random() * 2 - 1;
                this.color = ['#f43f5e', '#fb7185', '#fda4af', '#fecdd3'][Math.floor(Math.random() * 4)];
                this.opacity = Math.random() * 0.7 + 0.3;
            }

            update() {
                this.y += this.speedY;
                this.x += Math.sin(this.y * 0.01) + this.speedX;
                this.rotation += this.rotationSpeed;

                if (this.y > height + 20) {
                    this.reset();
                }
            }

            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate((this.rotation * Math.PI) / 180);
                ctx.globalAlpha = this.opacity;
                ctx.fillStyle = this.color;
                
                // Draw petal curve
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.bezierCurveTo(-this.size, -this.size / 2, -this.size, this.size, 0, this.size * 1.5);
                ctx.bezierCurveTo(this.size, this.size, this.size, -this.size / 2, 0, 0);
                ctx.fill();

                ctx.restore();
            }
        }

        const petals = Array.from({ length: 35 }, () => new Petal());

        function animatePetals() {
            ctx.clearRect(0, 0, width, height);
            petals.forEach(petal => {
                petal.update();
                petal.draw();
            });
            requestAnimationFrame(animatePetals);
        }
